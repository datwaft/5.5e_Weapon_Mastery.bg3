KBSECTION

  // Record the caster's Bonus Action value before the main-hand attack resolves.
  IF
    UsingSpell((CHARACTER)_Caster, "Target_MainHandAttack", _, _, _StoryActionID)
  AND
    HasPassive(_Caster, "WM55_Known_Nick", 1)
  AND
    GetActionResourceValuePersonal(_Caster, "BonusActionPoint", 0, _BonusActionPoints)
  THEN
    DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPoints);

  // Reasoning:
  //   WM55_NICK_READY is applied by the Nick passive during the main-hand attack.
  //   If the caster has spent a bonus action since the attack started, infer
  //   that BG3 performed its automatic off-hand attack. Refund the cost and
  //   record the automatic attack sequence.
  IF
    StatusApplied((CHARACTER)_Caster, "WM55_NICK_READY", _, _StoryActionID)
  AND
    DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPointsBefore)
  AND
    GetActionResourceValuePersonal(_Caster, "BonusActionPoint", 0, _BonusActionPointsAfter)
  AND
    _BonusActionPointsAfter < _BonusActionPointsBefore
  THEN
    ApplyStatus(_Caster, "WM55_NICK_AUTOMATIC_REFUND", 6.0, 1, _Caster);
    RemoveStatus(_Caster, "WM55_NICK_READY", _Caster);
    ApplyStatus(_Caster, "WM55_NICK_USED", 6.0, 1, _Caster);
    NOT DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPointsBefore);
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, 0);

  // Reasoning:
  //   If no Bonus Action was spent during the main-hand attack, BG3 did not
  //   perform its automatic off-hand attack. Clear the pending attack state and
  //   leave WM55_NICK_READY active for a manual off-hand attack.
  IF
    StatusApplied((CHARACTER)_Caster, "WM55_NICK_READY", _, _StoryActionID)
  AND
    DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPointsBefore)
  AND
    GetActionResourceValuePersonal(_Caster, "BonusActionPoint", 0, _BonusActionPointsAfter)
  AND
    _BonusActionPointsAfter >= _BonusActionPointsBefore
  THEN
    NOT DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPointsBefore);

  // Count the attacks that belong to the automatic dual-wield sequence. The
  // second attack is the off-hand attack, so block later off-hand attacks before
  // the action bar refreshes.
  IF
    AttackedBy(_, (CHARACTER)_AttackOwner, _, _, _, _, _StoryActionID)
  AND
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _AttackOwner, _CompletedAttacks)
  AND
    IntegerSum(_CompletedAttacks, 1, _NewCompletedAttacks)
  THEN
    NOT DB_WM55_Nick_AutomaticAttack(_StoryActionID, _AttackOwner, _CompletedAttacks);
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _AttackOwner, _NewCompletedAttacks);

  // After the automatic main-hand/off-hand sequence completes, block further
  // off-hand attacks unless Dual Wielder allows an additional one.
  IF
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks)
  AND
    _CompletedAttacks >= 2
  AND
    HasPassive(_Caster, "DualWielder_PassiveBonuses", 0)
  AND
    HasActiveStatus(_Caster, "WM55_NICK_OFFHAND_BLOCK", 0)
  THEN
    ApplyStatus(_Caster, "WM55_NICK_OFFHAND_BLOCK", 6.0, 1, _Caster);

  // CastedSpell is the fallback for misses and other sequences that do not emit
  // two AttackedBy events. These three cases are mutually exclusive, which lets
  // each case clean up its own automatic-attack fact safely.
  IF
    CastedSpell((CHARACTER)_Caster, "Target_MainHandAttack", _, _, _StoryActionID)
  AND
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks)
  AND
    HasPassive(_Caster, "DualWielder_PassiveBonuses", 0)
  AND
    HasActiveStatus(_Caster, "WM55_NICK_OFFHAND_BLOCK", 0)
  THEN
    ApplyStatus(_Caster, "WM55_NICK_OFFHAND_BLOCK", 6.0, 1, _Caster);
    NOT DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks);

  // If the off-hand block was already applied before the automatic sequence
  // finishes, only clean up the tracking state.
  IF
    CastedSpell((CHARACTER)_Caster, "Target_MainHandAttack", _, _, _StoryActionID)
  AND
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks)
  AND
    HasPassive(_Caster, "DualWielder_PassiveBonuses", 0)
  AND
    HasActiveStatus(_Caster, "WM55_NICK_OFFHAND_BLOCK", 1)
  THEN
    NOT DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks);

  // If the caster has Dual Wielder, only clean up the tracking state.
  IF
    CastedSpell((CHARACTER)_Caster, "Target_MainHandAttack", _, _, _StoryActionID)
  AND
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks)
  AND
    HasPassive(_Caster, "DualWielder_PassiveBonuses", 1)
  THEN
    NOT DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks);

  // Clean up an incomplete automatic-attack state when the caster's next turn begins.
  IF
    TurnStarted((CHARACTER)_Caster)
  AND
    DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPoints)
  THEN
    NOT DB_WM55_Nick_PendingMainHandAttack(_StoryActionID, _Caster, _BonusActionPoints);

  // Clean up an incomplete automatic-attack state when the caster's next turn begins.
  IF
    TurnStarted((CHARACTER)_Caster)
  AND
    DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks)
  THEN
    NOT DB_WM55_Nick_AutomaticAttack(_StoryActionID, _Caster, _CompletedAttacks);

EXITSECTION
